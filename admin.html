let lastScannedCode = ''; let lastScannedTime = 0; // VALIDASI KETAT GATE-OUT VS GATE-IN & ANTI DOUBLE-SCAN async function onScanSuccess(decodedText) { let code = decodedText.trim(); let now = Date.now(); // 1. FILTER COOLDOWN: Abaikan diam-diam jika QR yang sama ter-scan ulang dalam jeda < 2 detik if (code === lastScannedCode && (now - lastScannedTime) < 2000) { return; } // Update pencatatan scan terakhir lastScannedCode = code; lastScannedTime = now; // 2. ABAIKAN DIAM-DIAM jika kode sudah ada di daftar scan saat ini if (scannedPallets.includes(code)) { return; } let jenis = document.getElementById('jenisTransaksi').value; let isGateOut = jenis.includes("OUT"); // 3. Periksa status lokasi palet dari Supabase let { data: pData } = await db.from('master_pallet').select('status_lokasi').eq('pallet_id', code).single(); if (isGateOut) { // JIKA SCAN OUT: Palet WAJIB sedang berada di GUDANG if (pData && pData.status_lokasi && pData.status_lokasi.toUpperCase() !== 'GUDANG') { playBeep(true); showToast(`Gagal! Palet ${code} masih di ${pData.status_lokasi}. Wajib GATE-IN ke Gudang dulu!`, true); return; } } else { // JIKA SCAN IN: Peringatan jika palet memang sudah di Gudang if (pData && (!pData.status_lokasi || pData.status_lokasi.toUpperCase() === 'GUDANG')) { showToast(`Peringatan: Palet ${code} memang sudah berada di Gudang!`); } } // Masukkan ke daftar scan jika lolos validasi scannedPallets.unshift(code); playBeep(false); showToast("Palet " + code + " berhasil di-scan"); updateUI(); }
